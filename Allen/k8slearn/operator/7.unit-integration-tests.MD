### Step 2：编写单元测试

创建/更新 `internal/controller/tenant_controller_test.go`：

```go
package controller

import (
	"context"
	"testing"
	"time"

	. "github.com/onsi/ginkgo/v2"
	. "github.com/onsi/gomega"
	corev1 "k8s.io/api/core/v1"
	rbacv1 "k8s.io/api/rbac/v1"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"k8s.io/apimachinery/pkg/types"

	tenantv1alpha1 "github.com/cncamp/101/Allen/k8slearn/operator/tenant-operator/pkg/apis/tenant/v1alpha1"
)

var _ = Describe("Tenant Controller", func() {
	
	Context("when creating a new Tenant", func() {
		ctx := context.Background()
		tenantName := "test-tenant"
		tenantNamespace := "test-ns"

		It("should create a Namespace", func() {
			By("creating a Tenant CR")
			tenant := &tenantv1alpha1.Tenant{
				ObjectMeta: metav1.ObjectMeta{
					Name: tenantName,
				},
				Spec: tenantv1alpha1.TenantSpec{
					Namespace: tenantNamespace,
					AdminUser: "testuser",
				},
			}
			Expect(k8sClient.Create(ctx, tenant)).Should(Succeed())

			By("waiting for Namespace to be created")
			Eventually(func() error {
				ns := &corev1.Namespace{}
				return k8sClient.Get(ctx, types.NamespacedName{Name: tenantNamespace}, ns)
			}, timeout, interval).Should(Succeed())
		})

		It("should create a ResourceQuota", func() {
			By("checking ResourceQuota exists")
			Eventually(func() error {
				rq := &corev1.ResourceQuota{}
				return k8sClient.Get(ctx, types.NamespacedName{Name: "object-counts", Namespace: tenantNamespace}, rq)
			}, timeout, interval).Should(Succeed())
		})

		It("should create a RoleBinding", func() {
			By("checking RoleBinding exists")
			Eventually(func() error {
				rb := &rbacv1.RoleBinding{}
				return k8sClient.Get(ctx, types.NamespacedName{Name: "pod-admin", Namespace: tenantNamespace}, rb)
			}, timeout, interval).Should(Succeed())
		})

		It("should update Tenant status to Ready", func() {
			By("checking Tenant status")
			Eventually(func() string {
				tenant := &tenantv1alpha1.Tenant{}
				_ = k8sClient.Get(ctx, types.NamespacedName{Name: tenantName}, tenant)
				return tenant.Status.Phase
			}, timeout, interval).Should(Equal("Ready"))
		})
	})

	Context("when updating a Tenant", func() {
		ctx := context.Background()
		tenantName := "update-tenant"

		It("should update ResourceQuota when spec changes", func() {
			tenant := &tenantv1alpha1.Tenant{
				ObjectMeta: metav1.ObjectMeta{
					Name: tenantName,
				},
				Spec: tenantv1alpha1.TenantSpec{
					Namespace: "update-ns",
					ObjectCounts: &tenantv1alpha1.ObjectCountsQuota{
						ConfigMaps: "50",
					},
				},
			}
			Expect(k8sClient.Create(ctx, tenant)).Should(Succeed())

			Eventually(func() string {
				rq := &corev1.ResourceQuota{}
				_ = k8sClient.Get(ctx, types.NamespacedName{Name: "object-counts", Namespace: "update-ns"}, rq)
				return rq.Spec.Hard[corev1.ResourceConfigMaps].String()
			}, timeout, interval).Should(Equal("50"))
		})
	})

	Context("when deleting a Tenant", func() {
		ctx := context.Background()
		tenantName := "delete-tenant"

		It("should remove finalizer on deletion", func() {
			tenant := &tenantv1alpha1.Tenant{
				ObjectMeta: metav1.ObjectMeta{
					Name: tenantName,
				},
				Spec: tenantv1alpha1.TenantSpec{
					Namespace: "delete-ns",
				},
			}
			Expect(k8sClient.Create(ctx, tenant)).Should(Succeed())

			Expect(k8sClient.Delete(ctx, tenant)).Should(Succeed())

			Eventually(func() error {
				return k8sClient.Get(ctx, types.NamespacedName{Name: tenantName}, &tenantv1alpha1.Tenant{})
			}, timeout, interval).Should(SatisfyAny(BeNotFoundError(), BeNil()))
		})
	})
})
```

### Step 3：配置测试环境

更新 `internal/controller/suite_test.go`：

```go
package controller

import (
	"testing"

	. "github.com/onsi/ginkgo/v2"
	. "github.com/onsi/gomega"
	"k8s.io/client-go/kubernetes/scheme"
	"k8s.io/client-go/rest"
	"sigs.k8s.io/controller-runtime/pkg/client"
	"sigs.k8s.io/controller-runtime/pkg/envtest"
	"sigs.k8s.io/controller-runtime/pkg/envtest/printer"
	logf "sigs.k8s.io/controller-runtime/pkg/log"
	"sigs.k8s.io/controller-runtime/pkg/log/zap"

	tenantv1alpha1 "github.com/cncamp/101/Allen/k8slearn/operator/tenant-operator/pkg/apis/tenant/v1alpha1"
)

var cfg *rest.Config
var k8sClient client.Client
var testEnv *envtest.Environment

func TestControllers(t *testing.T) {
	RegisterFailHandler(Fail)
	RunSpecsWithDefaultAndCustomReporters(t, "Controller Suite", []Reporter{printer.NewlineReporter{}})
}

var _ = BeforeSuite(func() {
	logf.SetLogger(zap.New(zap.WriteTo(GinkgoWriter)))

	By("bootstrapping test environment")
	testEnv = &envtest.Environment{
		CRDDirectoryPaths: []string{
			filepath.Join("..", "..", "config", "crd", "bases"),
		},
		ErrorIfCRDPathMissing: true,
	}

	var err error
	cfg, err = testEnv.Start()
	Expect(err).ToNot(HaveOccurred())
	Expect(cfg).ToNot(BeNil())

	err = tenantv1alpha1.AddToScheme(scheme.Scheme)
	Expect(err).NotTo(HaveOccurred())

	k8sClient, err = client.New(cfg, client.Options{Scheme: scheme.Scheme})
	Expect(err).ToNot(HaveOccurred())
	Expect(k8sClient).ToNot(BeNil())
})

var _ = AfterSuite(func() {
	By("tearing down the test environment")
	err := testEnv.Stop()
	Expect(err).ToNot(HaveOccurred())
})
```

### 常用测试命令

| 命令 | 说明 |
|------|------|
| `make test` | 运行所有测试 |
| `make test V=1` | 显示详细输出 |
| `ginkgo -v` | 运行测试并显示详情 |
| `ginkgo --focus="should create"` | 只运行匹配用例 |
